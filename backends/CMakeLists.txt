cmake_minimum_required(VERSION 3.25)
if(NOT TARGET strata_kernels OR NOT TARGET strata_spec)
  message(FATAL_ERROR "Configure the reviewed Strata root with the optional-backend injection")
endif()
if(NOT STRATA_ISA_FLOOR STREQUAL "avx" OR STRATA_PORTABLE OR NOT CMAKE_CUDA_ARCHITECTURES STREQUAL "61")
  message(FATAL_ERROR "This qualification recipe requires the AVX floor and sm_61")
endif()
set(CMAKE_CXX_STANDARD 20)
enable_testing()
set(P "${CMAKE_CURRENT_LIST_DIR}/poly")
set(G "${CMAKE_BINARY_DIR}/ggml-pascallama-source/ggml")
if(NOT EXISTS "${G}/src/ggml-cuda/mmq.cuh")
  message(FATAL_ERROR "The reviewed Pascal overlay build copy is required")
endif()
set(M "${G}/src/ggml-cuda")
set(instances)
foreach(t q4_0 q4_1 q5_0 q5_1 q8_0 q2_k q3_k q4_k q5_k q6_k iq1_s iq2_xxs iq2_xs iq2_s iq3_xxs iq3_s iq4_nl iq4_xs mxfp4)
  list(APPEND instances "${M}/template-instances/mmq-instance-${t}.cu")
endforeach()
add_library(strata_poly_mmq STATIC "${P}/src/poly/mmq.cu" "${P}/src/poly/fattn.cu"
  "${STRATA_CORE_SOURCE}/src/prefill/ggml_cuda_host.cu" "${M}/quantize.cu" ${instances})
target_include_directories(strata_poly_mmq PRIVATE "${G}/include" "${G}/src" "${M}"
  "${STRATA_CORE_SOURCE}/include" "${P}/include")
target_compile_options(strata_poly_mmq PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda;-Xcompiler=/bigobj>)
target_link_libraries(strata_poly_mmq PUBLIC ggml-base CUDA::cudart)
set(engine_files main.cpp engine.cpp glm5next.cpp qwen35.cpp gemma4.cpp deepseek4.cpp lfm2moe.cpp flint.cpp cpu_experts.cpp kernels.cu kernels_dsv4.cu)
list(TRANSFORM engine_files PREPEND "${P}/src/poly/")
add_executable(strata-poly ${engine_files} "${STRATA_CORE_SOURCE}/src/core/conversation_memory.cpp")
target_include_directories(strata-poly PRIVATE "${STRATA_CORE_SOURCE}/include" "${P}/include" ${CUDAToolkit_INCLUDE_DIRS})
target_compile_options(strata-poly PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda;-Xcompiler=/bigobj>)
target_link_libraries(strata-poly PRIVATE strata_poly_mmq strata_kernels strata_spec ggml-cpu ggml-base CUDA::cublas CUDA::cudart CUDA::cuda_driver Threads::Threads)
add_executable(poly_workers_test "${P}/src/poly/workers_test.cpp" "${P}/src/poly/cpu_experts.cpp")
target_include_directories(poly_workers_test PRIVATE "${G}/include")
target_link_libraries(poly_workers_test PRIVATE ggml-cpu ggml-base Threads::Threads)
add_test(NAME poly_workers COMMAND poly_workers_test)
add_executable(poly_guesses_test "${P}/src/poly/guesses_test.cpp")
target_include_directories(poly_guesses_test PRIVATE "${STRATA_CORE_SOURCE}/include")
target_link_libraries(poly_guesses_test PRIVATE strata_spec)
add_test(NAME poly_guesses COMMAND poly_guesses_test)
add_executable(poly_conversations_test "${P}/src/poly/conversations_test.cpp")
add_test(NAME poly_conversations COMMAND poly_conversations_test)
