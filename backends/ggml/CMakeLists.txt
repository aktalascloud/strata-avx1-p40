cmake_minimum_required(VERSION 3.25)
project(strata_ggml_backend LANGUAGES C CXX)
set(STRATA_DEVICE "cpu" CACHE STRING "cpu, vulkan or cuda; no standalone llama.cpp server is built")
if(NOT STRATA_DEVICE MATCHES "^(cpu|vulkan|cuda)$")
  message(FATAL_ERROR "Unknown embedded backend")
endif()
if(NOT EXISTS "${STRATA_LLAMA_SOURCE}/include/llama.h")
  message(FATAL_ERROR "A pinned local llama.cpp source tree is required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
if(STRATA_DEVICE STREQUAL "cuda")
  if(NOT CMAKE_CUDA_ARCHITECTURES STREQUAL "61")
    message(FATAL_ERROR "This experimental CUDA recipe requires the reviewed sm61 target")
  endif()
  enable_language(CUDA)
  set(GGML_CUDA ON CACHE BOOL "" FORCE)
else()
  set(GGML_CUDA OFF CACHE BOOL "" FORCE)
endif()
set(GGML_HIP OFF CACHE BOOL "" FORCE)
set(GGML_SYCL OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_AVX ON CACHE BOOL "" FORCE)
foreach(flag AVX2 FMA F16C AVX512 AVX512_VBMI AVX512_VNNI AVX512_BF16 AMX_TILE AMX_INT8 AMX_BF16)
  set(GGML_${flag} OFF CACHE BOOL "" FORCE)
endforeach()
if(STRATA_DEVICE STREQUAL "vulkan")
  set(GGML_VULKAN ON CACHE BOOL "" FORCE)
else()
  set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
endif()
add_subdirectory("${STRATA_LLAMA_SOURCE}" dependency EXCLUDE_FROM_ALL)
add_executable(strata-ggml main.cpp)
target_compile_definitions(strata-ggml PRIVATE STRATA_DEVICE_NAME="${STRATA_DEVICE}")
if(EXISTS "${STRATA_LLAMA_SOURCE}/src/models/glm5-next.cpp")
  target_compile_definitions(strata-ggml PRIVATE STRATA_GLM5_NEXT=1)
endif()
if(STRATA_LLAMA_PIN)
  target_compile_definitions(strata-ggml PRIVATE STRATA_LLAMA_PIN="${STRATA_LLAMA_PIN}")
endif()
target_link_libraries(strata-ggml PRIVATE llama-common llama ${CMAKE_THREAD_LIBS_INIT})
file(READ "${STRATA_LLAMA_SOURCE}/common/common.h" _strata_common_header)
if(_strata_common_header MATCHES "llama_batch_ext_ptr batch;")
  target_compile_definitions(strata-ggml PRIVATE STRATA_BATCH_EXT=1)
endif()
